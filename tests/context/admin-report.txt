  function normalizeText(value?: unknown) {
    if (value === null || value === undefined) {
      return undefined;
    }

    if (typeof value === "string") {
      const trimmedValue = value.trim();
      return trimmedValue ? trimmedValue : undefined;
    }

    if (typeof value === "number" || typeof value === "boolean" || typeof value === "bigint") {
      return String(value);
    }

    if (dayjs.isDayjs(value)) {
      return value.format("YYYY-MM-DD");
    }

    return undefined;
  }

  function normalizeNumber(value?: unknown) {
    if (value === null || value === undefined || value === "") {
      return undefined;
    }

    if (typeof value === "number") {
      return Number.isFinite(value) ? value : undefined;
    }

    const normalizedValue = normalizeText(value);
    if (!normalizedValue) {
      return undefined;
    }

    const parsedValue = Number(normalizedValue);
    return Number.isFinite(parsedValue) ? parsedValue : undefined;
  }

  function normalizeStringArray(value?: unknown) {
    if (Array.isArray(value)) {
      const normalizedValues = value
        .map((item) => normalizeText(item))
        .filter((item): item is string => !!item);

      return normalizedValues.length > 0 ? normalizedValues : undefined;
    }

    const normalizedValue = normalizeText(value);
    if (!normalizedValue) {
      return undefined;
    }

    const values = normalizedValue
      .split(",")
      .map((item) => item.trim())
      .filter(Boolean);

    return values.length > 0 ? values : undefined;
  }

  const parsePartialThaiDate = (
    value?: string | null,
    type: PartialDateType = "DATE",
  ) => {
    if (!value) return null;

    const normalizedValue = String(value).trim();
    const adjustYear = (yearText: string) => {
      const yearNumber = Number(yearText);
      return Number.isFinite(yearNumber) && yearNumber >= 2400
        ? String(yearNumber - 543)
        : yearText;
    };

    if (type === "YEAR" && /^\d{4}$/.test(normalizedValue)) {
      const parsed = dayjs(`${adjustYear(normalizedValue)}-01-01`);
      return parsed.isValid() ? parsed : null;
    }

    if (type === "MONTH") {
      const match = normalizedValue.match(/^(\d{4})-(\d{2})$/);
      if (match) {
        const parsed = dayjs(`${adjustYear(match[1])}-${match[2]}-01`);
        return parsed.isValid() ? parsed : null;
      }
    }

    const dateMatch = normalizedValue.match(/^(\d{4})-(\d{2})-(\d{2})$/);
    if (dateMatch) {
      const parsed = dayjs(
        `${adjustYear(dateMatch[1])}-${dateMatch[2]}-${dateMatch[3]}`,
      );
      return parsed.isValid() ? parsed : null;
    }

    const parsed = dayjs(normalizedValue);
    return parsed.isValid() ? parsed : null;
  };

  function parseDescriptionMap(value?: unknown, selectedValues: string[] = []) {
    const normalizedValue = normalizeText(value);
    if (!normalizedValue) {
      return {} as Record<string, string>;
    }

    try {
      const parsedValue = JSON.parse(normalizedValue) as Record<string, unknown>;
      return Object.entries(parsedValue).reduce<Record<string, string>>((result, [key, item]) => {
        const normalizedItem = normalizeText(item);

        if (normalizedItem && (selectedValues.length === 0 || selectedValues.includes(key))) {
          result[key] = normalizedItem;
        }

        return result;
      }, {});
    } catch {
      if (selectedValues.length === 1) {
        return { [selectedValues[0]]: normalizedValue };
      }

      return {} as Record<string, string>;
    }
  }

  function serializeDescriptionMap(value: Record<string, string>) {
    const normalizedEntries = Object.entries(value).reduce<Record<string, string>>((result, [key, item]) => {
      const normalizedItem = normalizeText(item);

      if (normalizedItem) {
        result[key] = normalizedItem;
      }

      return result;
    }, {});

    return Object.keys(normalizedEntries).length > 0 ? JSON.stringify(normalizedEntries) : undefined;
  }

  const getReportDateValue = (record: Report) => {
    if (record.datasetGroupsDate) {
      const match = record.datasetGroupsDate.match(
        /^(\d{4})-(\d{2})-(\d{2})(?:[T\s](\d{2}):(\d{2})(?::(\d{2}))?)?$/,
      );

      if (match) {
        const sourceYear = Number(match[1]);
        const gregorianYear = sourceYear > 2400 ? sourceYear - 543 : sourceYear;

        return dayjs(
          new Date(
            gregorianYear,
            Number(match[2]) - 1,
            Number(match[3]),
            Number(match[4] ?? "0"),
            Number(match[5] ?? "0"),
            Number(match[6] ?? "0"),
          ),
        );
      }

      return dayjs(record.datasetGroupsDate);
    }

    if (record.yearCode && record.monthCode) {
      const adYear =
        parseInt(record.yearCode) > 2500
          ? parseInt(record.yearCode) - 543
          : parseInt(record.yearCode);

      return dayjs(new Date(adYear, parseInt(record.monthCode) - 1, 1));
    }

    return null;
  };

  const buildMetadataDetailOptions = (
    metadata?: ReportDetailMetadata,
  ): Partial<Record<number, SelectOption>> => {
    if (!metadata) return {};

    return METADATA_SELECT_FIELD_CONFIGS.reduce<Partial<Record<number, SelectOption>>>(
      (result, config) => {
        const label = normalizeText(metadata[config.labelField]);
        const detailFieldConfig = METADATA_SELECT_DETAIL_FIELD_CONFIGS[config.metadataId];
        const fallbackValue = normalizeText(metadata[config.valueField]);
        const code = detailFieldConfig?.codeField
          ? normalizeText(metadata[detailFieldConfig.codeField])
          : fallbackValue;
        const metadataListId = detailFieldConfig?.listIdField
          ? (metadata[detailFieldConfig.listIdField] as number | null | undefined)
          : undefined;
        const otherValue = detailFieldConfig?.otherField
          ? metadata[detailFieldConfig.otherField]
          : undefined;
        const value = detailFieldConfig
          ? getSingleSelectValueFromDetail(
            config.metadataId,
            code ?? fallbackValue,
            metadataListId,
            otherValue,
          )
          : fallbackValue;

        if (value && label) {
          result[config.metadataId] = {
            value,
            label,
            code,
          };
        }

        return result;
      },
      {},
    );
  };

  const buildDictionaryRowsFromDetail = (
    dictionaryItems?: ReportDetailDictionaryItem[],
  ): DataDictionaryRow[] => {
    if (!Array.isArray(dictionaryItems) || dictionaryItems.length === 0) {
      return [createEmptyDictionaryRow(1)];
    }

    return dictionaryItems.map((item, index): DataDictionaryRow => ({
      id: item.dictId ?? index + 1,
      columnName: item.columnName ?? "",
      dataType:
        item.dataTypeId === null || item.dataTypeId === undefined
          ? item.dataType ?? ""
          : String(item.dataTypeId),
      sizeValue:
        item.sizeValue === null || item.sizeValue === undefined
          ? ""
          : String(item.sizeValue),
      description: item.description ?? "",
      required: normalizeRequired(item.required),
      sampleData: item.sampleData ?? "",
    }));
  };

  const getSingleSelectValueFromDetail = (
    metadataId: number,
    code?: unknown,
    metadataListId?: number | null,
    otherValue?: unknown,
  ) => {
    const normalizedCode = normalizeText(code);
    const normalizedOtherValue = normalizeText(otherValue);
    const otherCode = OTHER_METADATA_CODES[metadataId];
    const options = getMetadataOptions(metadataId);

    // ใช้ metadataListId เฉพาะเมื่อมี option นี้อยู่จริง
    if (metadataListId != null) {
      const normalizedListId = String(metadataListId);

      if (options.some((option) => option.value === normalizedListId)) {
        return normalizedListId;
      }
    }

    // ถ้าเป็น other ให้ select แสดง option หลัก “อื่น ๆ”
    // ส่วน input ให้ใช้ otherValue แยกต่างหาก
    if (
      normalizedCode &&
      otherCode &&
      normalizedCode === otherCode &&
      normalizedOtherValue
    ) {
      return getOtherOptionValue(metadataId);
    }

    // พยายามหา option จาก code หรือ value (code ถูก normalize แล้วใน getMetadataOptions)
    if (normalizedCode) {
      const matchedOption = options.find(
        (option) => option.code === normalizedCode || option.value === normalizedCode,
      );

      if (matchedOption) {
        return matchedOption.value;
      }
    }

    // ถ้าหาไม่เจอเลย คืน undefined แทนที่จะคืน code เพื่อให้ form แสดง validation
    return undefined;
  };

                                filterOption={(input, option) =>
                                  String(option?.label ?? "").replace(/\s+/g, "").toLowerCase().includes(input.replace(/\s+/g, "").toLowerCase())
                                }
                                onChange={(value: FlexibleDateType) =>
                                  setMetaData((prev) => ({
                                    ...prev,
                                    dataPublishedDateType: value,
                                    dataPublishedDate:
                                      value === "DATETIME" ? null : prev.dataPublishedDate,
                                    dataPublishedDateText:
                                      value === "TEXT" ? prev.dataPublishedDateText : undefined,
                                  }))
                                }
                              />
                            </Form.Item>

                            {(metaData.dataPublishedDateType ?? "DATETIME") === "DATETIME" ? (
                              <Form.Item name="dataPublishedDate" noStyle>
                                <ThaiDateTimePicker
                                  id="admin-report-data-published-date"
                                  value={
                                    dayjs.isDayjs(metaData.dataPublishedDate)
                                      ? metaData.dataPublishedDate
                                      : metaData.dataPublishedDate
                                        ? parseFlexibleDateTimeValue(
                                          String(metaData.dataPublishedDate),
                                          metaData.dataPublishedDateType,
                                        )
                                        : null
                                  }
                                  onChange={(date: Dayjs | Dayjs[] | null) =>
                                    setMetaData((prev) => ({
                                      ...prev,
                                      dataPublishedDate: Array.isArray(date)
                                        ? date[0] ?? null
                                        : date,
                                      dataPublishedDateText: undefined,
                                    }))
                                  }
                                  placeholder="วันที่เผยแพร่ข้อมูล"
                                  displayFormat="BBBB-MM-DD-HH-mm"
                                  timeFormat="HH-mm"
                                  format={(value) =>
                                    formatFlexibleDateForDisplay(value, metaData.dataPublishedDateType) ?? ""
                                  }
                                />
                              </Form.Item>
                            ) : (
                              <Form.Item name="dataPublishedDateText" noStyle>
                                <Input.TextArea
                                  id="admin-report-data-published-date-text"
                                  placeholder="เช่น เผยแพร่ทุกไตรมาสหลังปิดงวดข้อมูล หรือ ทุกวันที่ 1 ของเดือน เวลา 09.00 น."
                                  autoSize={{ minRows: 3, maxRows: 5 }}
                                  value={metaData.dataPublishedDateText ?? ""}
                                  onChange={(e) =>
                                    setMetaData((prev) => ({
                                      ...prev,
                                      dataPublishedDate: null,
                                      dataPublishedDateText: e.target.value,
                                    }))
                                  }
                                />
                              </Form.Item>
                            )}
                          </div>
                        </Form.Item>
                      </>
                    )}
                  </>
                )}

                {step === 3 && (
                  <div className="col-span-1 md:col-span-2 xl:col-span-2 space-y-4">
                    <div className="flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-between">
                      <div>
                        <div className="text-base font-semibold text-black/85">
                          ข้อมูล Data Dictionary
                        </div>
                        <div className="text-sm text-black/55">
                          เพิ่มรายละเอียดคอลัมน์ข้อมูลโดยอ้างอิงรูปแบบตารางจากหน้า InfoTable
                        </div>
                      </div>

                      <Button
                        onClick={addDataDictionaryRow}
                        id="admin-report-add-dict-row"
                        className="flex items-center gap-2 bg-sky-600! hover:bg-sky-700! border-sky-600! hover:border-sky-700! text-white!"
                      >
                        <Plus size={16} />
                        เพิ่มแถว
                      </Button>
                    </div>

                    <DataTable<DataDictionaryRow>
                      showSearch={false}
                      showAddButton={false}
                      columns={dataDictionaryColumns}
                      dataSource={dataDictionaryRows}
                      pagination={false}
                      rowKey="id"
                    />
                  </div>
                )}

                <div className="col-span-1 md:col-span-2 xl:col-span-2 flex flex-wrap justify-end gap-3 border-t border-slate-200 pt-6">

                  {step === 1 && (
                    <>
                      {/* ปุ่มควบคุมของ step 1 */}
                      <Button
                        id="admin-report-step-1-cancel"
                        disabled={false}
                        onClick={resetEditor}
                        className="bg-(--cancel-btn)! border-(--cancel-btn)! hover:bg-(--cancel-btn-hover)! text-white!"
                      >
                        ยกเลิก
                      </Button>
                      <Button
                        id="admin-report-step-1-next"
                        disabled={false}
                        // onClick={isViewMode ? () => setStep(2) : goToStepTwo}
                        onClick={async () => {
                          if (isViewMode) {
                            setStep(3);
                            return;
                          }

                          try {
                            await editorForm.validateFields();
                          } catch {
                            scrollToFirstValidationError();
                            message.error("กรุณากรอกข้อมูลให้ครบถ้วน");
                            return;
                          }

                          setStep(2);
                        }}
                        className="flex items-center gap-2 bg-[#059669]! hover:bg-[#047857]! border-emerald-600! hover:border-emerald-700! text-white!"
                      >
                        ถัดไป
                      </Button>
                    </>
                  )}

                  {step === 2 && (
                    <>
                      {/* ปุ่มควบคุมของ step 2 และขั้นตอนบันทึกข้อมูล */}
                      <Button
                        id="admin-report-step-2-back"
                        disabled={false}
                        onClick={() => setStep(1)}
                        className="bg-gray-500! border-gray-500! hover:bg-gray-600! hover:border-gray-600! text-white!"
                      >
                        ย้อนกลับ
                      </Button>

                      <Button
                        id="admin-report-step-2-next"
                        disabled={false}
                        onClick={async () => {
                          if (isViewMode) {
                            setStep(3);
                            return;
                          }

                          try {
                            await editorForm.validateFields();
                          } catch {
                            scrollToFirstValidationError();
                            message.error("กรุณาเลือกข้อมูล Meta Data ให้ครบถ้วน");
                            return;
                          }

                          setStep(3);
                        }}
                        className="flex items-center gap-2 bg-[#059669]! hover:bg-[#047857]! border-emerald-600! hover:border-emerald-700! text-white!"
                      >
                        ถัดไป
                      </Button>
                    </>
                  )}

                  {step === 3 && (
                    <>
                      <Button
                        disabled={false}
                        onClick={() => setStep(2)}
                        className="bg-gray-500! border-gray-500! hover:bg-gray-600! hover:border-gray-600! text-white!"
                      >
                        ย้อนกลับ
                      </Button>

                      {isViewMode ? (
                        <Button
                          disabled={false}
                          onClick={resetEditor}
                          className="flex items-center gap-2 bg-[#0f60d8]! hover:bg-[#0b4fb2]! border-[#0f60d8]! hover:border-[#0b4fb2]! text-white!"
                        >
                          ปิด
                        </Button>
                      ) : (
                        <Button
                          id="admin-report-save"
                          disabled={isSaving}
                          loading={isSaving}
                          onClick={async () => {
                            if (!validateDataDictionaryStep()) {
                              message.error("กรุณากรอก Column Name และ Data Type ให้ครบในแถวที่มีข้อมูล");
                              return;
                            }

                            if (editTicket) {
                              await handleSaveReportEdit();
                              return;
                            }

                            await handleCreateReport();
                          }}
                          className="flex items-center gap-2 bg-[#059669]! hover:bg-[#047857]! border-emerald-600! hover:border-emerald-700! text-white!"
                        >
                          <Plus size={16} />
                          บันทึก
                        </Button>
                      )}
                    </>
                  )}

                </div>
              </div>

            </div>
          </div>
        </Form>

      </div>


      {/* ส่วนตารางแสดงรายการรายงาน */}
      <DataTable<Report>
        headerWrapperClassName="bg-linear-to-r from-violet-600/15 via-sky-400/10 to-transparent px-5 py-3"
        headerTitle={(
          <Form layout="vertical" component={false}>
            <div className="grid grid-cols-1 md:grid-cols-2 xl:grid-cols-3 gap-6">

              <FieldItem label="กลุ่มข้อมูลรายงาน">
                <AntSelect value={main} onChange={setMain} options={[{ value: "all", label: "กลุ่มข้อมูลรายงานทั้งหมด" }, ...mainOptions]} />
              </FieldItem>

              <FieldItem label="กลุ่มรายงาน">
                <AntSelect value={sub} onChange={setSub} options={[{ value: "all", label: "กลุ่มรายงานทั้งหมด" }, ...subOptions]} />
              </FieldItem>

              <FieldItem label="ชุดข้อมูลรายงาน">
                <AntSelect value={category} onChange={setCategory} options={[{ value: "all", label: "ชุดข้อมูลรายงานทั้งหมด" }, ...categoryOptions]} />
              </FieldItem>

              <FieldItem label="สถานะ">
                <AntSelect value={status} onChange={setStatus} options={STATUS_OPTIONS} />
              </FieldItem>

              <Space>
                <Button
                  onClick={handleReset}
                  className="bg-(--reset-btn)! text-white! border-(--reset-btn)! hover:bg-(--reset-btn-hover)! hover:border-(--reset-btn-hover)! hover:text-white!"
                >
                  ล้างข้อมูล
                </Button>
              </Space>

              <FieldItem label="ค้นหาชื่อรายงาน">
                <Input
                  id="admin-report-search-name"
                  value={search}
                  prefix={<Search size={16} className="text-gray-400" />}
                  onChange={(e) => setSearch(e.target.value)}
                  placeholder="ค้นหาชื่อรายงาน..."
                  allowClear
                  className="w-64 rounded-lg"
                />
              </FieldItem>
            </div>
          </Form>
        )}

        columns={columns}
        showSearch={false}
        showAddButton={false}
        addButtonText="เพิ่มรายงาน"
        onAdd={() => {
          resetEditor();
        }}
        onSearch={(value) => setSearch(value)}
        dataSource={data}
        loading={loading}
        rowKey="id"
        pagination={{
          defaultPageSize: 10,
          showSizeChanger: true,
          pageSizeOptions: ["10", "20", "50", "100"],
        }}
      />
    </div>
  );
}
